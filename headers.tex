%% Cette section prend un header name (le nom de la pièce),
%% assigne son label (le nom de fichier) à la pièce, et un nom au label (le nom de la pièce),
%% et mémorise le nom dans \namebuffer pour indexation future une fois qu'on a le mode.
\begin{luacode*}
function delete_newline ( s )
   s = string.gsub ( s, 'newline', '')
   s = string.gsub ( s, 'hspace', '')
   s = string.gsub ( s, '4mm', '')
   s = string.gsub ( s, 'protect', '')
   s = string.gsub ( s, 'hbox', '')
   s = string.gsub ( s, [[\]], '')
   s = string.gsub ( s, "{}", '')
   tex.sprint ( s )
end
\end{luacode*}

\begin{luacode*}
local glyph = node.id("glyph")
local glue  = node.id("glue")
local kern  = node.id("kern")

local function fix_space_T(head)
  local n = head
  while n do
    if n.id == glue then
      local next = n.next
      if next and next.id == glyph then
        local char = unicode.utf8.char(next.char)
        if char == "T" or char == "V" then
          -- Création du kern
          local k = node.new(kern)
          k.kern = 1.5 * 65536  -- 1pt

          -- Insertion du kern entre l'espace et le T
          k.next = next
          n.next = k
        end
      end
    end
    n = n.next
  end
  return head
end

luatexbase.add_to_callback("pre_linebreak_filter", fix_space_T, "fix_space_T")
luatexbase.add_to_callback("hpack_filter", fix_space_T, "fix_space_T_hpack")
\end{luacode*}
