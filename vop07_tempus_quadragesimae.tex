%%%% INFRA HEBDOMADAM I QUADRAGRESIMAE
\festum{}{Feria secunda}{}{Infra hebdomadam I Quadragesimæ}{3}
\cantus{Q1F2VAM}{Ad Magn.}
\oratio{Absólve, quǽsumus, Dómine, nostrórum víncula peccatórum;\pscross{}et, quidquid pro eis merémur, propitiátus avérte.\psstar{} Per.}

\festum{}{Feria tertia}{}{Infra hebdomadam I Quadragesimæ}{3}
\cantus{Q1F3VAM}{Ad Magn.}
\oratio{Ascéndant ad te, Dómine, preces nostræ;\pscross{}et ab Ecclésia tua cunctam repélle nequítiam.\psstar{} Per.}

\festum{}{Feria quarta Quatuor Temporum}{}{Infra hebdomadam I Quadragesimæ}{3}
\cantus{Q1F4VAM}{Ad Magn.}
\oratio{Mentes nostras, quǽsumus, Dómine, lúmine tuæ claritátis illústra;\pscross{}ut vidére possímus quæ agénda sunt, et quæ recta sunt ágere valeámus.\psstar{} Per.}

\festum{}{Feria quinta}{}{Infra hebdomadam I Quadragesimæ}{3}
\cantus{Q1F5VAM}{Ad Magn.}
\oratio{Da, quǽsumus, Dómine, pópulis christiánis et quæ profiténtur agnóscere;\pscross{}et cæléste munus dilígere, quod frequéntant.\psstar{} Per.}

\festum{}{Feria sexta Quatuor Temporum}{}{Infra hebdomadam I Quadragesimæ}{3}
\cantus{Q1F6VAM}{Ad Magn.}
\oratio{Exáudi nos, miséricors Deus,\pscross{}et méntibus nostris grátiæ tuæ lumen osténde.\psstar{} Per.}

%%%% INFRA HEBDOMADAM II QUADRAGRESIMAE
\festum{}{Feria secunda}{}{Infra hebdomadam II Quadragesimæ}{3}
\cantus{Q2F2VAM}{Ad Magn.}
\oratio{Adésto supplicatiónibus nostris, omnípotens Deus:\psstar{} et, quibus fidúciam sperándæ pietátis indúlges,\pscross{}consuétæ misericórdiæ tríbue benígnus efféctum.\psstar{} Per.}

\festum{}{Feria tertia}{}{Infra hebdomadam II Quadragesimæ}{3}
\cantus{Q2F3VAM}{Ad Magn.}
\oratio{Propitiáre, Dómine, supplicatiónibus nostris:\psstar{} et animárum nostrárum medére languóribus;\pscross{}ut, remissióne percépta, in tua semper benedictióne lætémur.\psstar{} Per.}

\festum{}{Feria quarta}{}{Infra hebdomadam II Quadragesimæ}{3}
\cantus{Q2F4VAM}{Ad Magn.}
\oratio{Deus, innocéntiæ restitútor et amátor, dírige ad te tuórum corda servórum:\psstar{} ut, spíritus tui fervóre concépto, et in fide inveniántur stábiles\pscross{}et in ópere efficáces.\psstar{} Per.}

\festum{}{Feria quinta}{}{Infra hebdomadam II Quadragesimæ}{3}
\cantus{Q2F5VAM}{Ad Magn.}
\oratio{Adésto, Dómine, fámulis tuis,\pscross{}et perpétuam benignitátem largíre poscéntibus:\psstar{} ut iis, qui te auctóre et gubernatóre gloriántur, et congregáta restáures,\pscross{}et restauráta consérves.\psstar{} Per.}

\festum{}{Feria sexta}{}{Infra hebdomadam II Quadragesimæ}{3}
\cantus{Q2F6VAM}{Ad Magn.}
\oratio{Da, quǽsumus, Dómine, pópulo tuo salútem mentis et córporis:\psstar{} ut, bonis opéribus inhæréndo,\pscross{}tuæ semper virtútis mereátur protectióne deféndi.\psstar{} Per.}
